```rust
pub async fn get_chunk(
    &self,
    date: NaiveDate,
) -> Result<Arc<ChunkData>, ChunkManagerError> {
    let key = ChunkKey { date };
    
    // 1. Check cache first
    {
        let mut cache = self.cache.write().await;
        if let Some(data) = cache.get(&key) {
            return Ok(Arc::clone(data));
        }
    }
    
    // 2. Cache miss - load from database
    let chunk_data = self.load_chunk_from_db(date).await?;
    let chunk_arc = Arc::new(chunk_data);
    
    // 3. Populate cache
    {
        let mut cache = self.cache.write().await;
        cache.put(key, Arc::clone(&chunk_arc));
    }
    
    Ok(chunk_arc)
}
```

5. **Database loading method**:
```rust
async fn load_chunk_from_db(
    &self,
    date: NaiveDate,
) -> Result<ChunkData, ChunkManagerError> {
    let start_time = date.and_hms_opt(0, 0, 0).unwrap().and_utc();
    let end_time = date.and_hms_opt(23, 59, 59).unwrap().and_utc();
    
    // Load planet positions
    let positions = sqlx::query_as::<_, schema::PlanetPosition>(
        r#"
        SELECT time, body_id, longitude, latitude, distance, speed_lon, retrograde, zodiac_sign
        FROM planet_positions
        WHERE time >= $1 AND time <= $2
        ORDER BY time, body_id
        "#
    )
    .bind(start_time)
    .bind(end_time)
    .fetch_all(self.db_pool.pool())
    .await?;
    
    // Load aspects
    let aspects = sqlx::query_as::<_, schema::Aspect>(
        r#"
        SELECT time, body1_id, body2_id, aspect_type, orb, applying
        FROM aspects
        WHERE time >= $1 AND time <= $2
        ORDER BY time
        "#
    )
    .bind(start_time)
    .bind(end_time)
    .fetch_all(self.db_pool.pool())
    .await?;
    
    // Load lunar conditions
    let lunar = sqlx::query_as::<_, schema::LunarCondition>(
        r#"
        SELECT time, moon_phase, moon_sign, moon_phase_angle, moon_illumination,
               is_void_of_course, voc_start, voc_end
        FROM lunar_conditions
        WHERE time >= $1 AND time <= $2
        ORDER BY time
        "#
    )
    .bind(start_time)
    .bind(end_time)
    .fetch_all(self.db_pool.pool())
    .await?;
    
    // Convert to compact format
    let compact_positions = positions.iter()
        .map(|p| CompactPlanetPosition::from_schema(p, date))
        .collect::<Result<Vec<_>, _>>()?;
    
    let compact_aspects = aspects.iter()
        .map(|a| CompactAspect::from_schema(a, date))
        .collect::<Result<Vec<_>, _>>()?;
    
    let compact_lunar = lunar.iter()
        .map(|l| CompactLunarCondition::from_schema(l, date))
        .collect::<Result<Vec<_>, _>>()?;
    
    Ok(ChunkData {
        date,
        planet_positions: compact_positions,
        aspects: compact_aspects,
        lunar_conditions: compact_lunar,
        loaded_at: Instant::now(),
    })
}
```

pub async fn cache_stats(&self) -> (usize, usize) {

pub async fn clear_cache(&self) {

pub async fn is_cached(&self, date: NaiveDate) -> bool {

impl CompactPlanetPosition {
    pub fn from_schema(
        pos: &crate::database::schema::PlanetPosition,
        chunk_date: NaiveDate,
    ) -> Result<Self, ChunkError> {
        let chunk_start = chunk_date.and_hms_opt(0, 0, 0).unwrap().and_utc();
        let minutes_since_start = (pos.time - chunk_start).num_minutes() as u32;
        
        // Convert Decimal to millidegrees
        let longitude_milli = (pos.longitude * Decimal::from(1000))
            .to_i32()
            .ok_or(ChunkError::Conversion("longitude".to_string()))?;
        
        let speed_milli = (pos.speed_lon * Decimal::from(1000))
            .to_i16()
            .ok_or(ChunkError::Conversion("speed".to_string()))?;
        
        Ok(Self {
            timestamp_minutes: minutes_since_start,
            longitude_millidegrees: longitude_milli,
            speed_millidegrees: speed_milli,
            body_id: pos.body_id as u8,
            zodiac_sign: pos.zodiac_sign as u8,
